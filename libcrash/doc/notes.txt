/*
 * page_compress explanation:
 * 
 * page compression implementation is broken into 2 stages
 * 
 * stage 1)
 * the first stage runs/loops for each byte of input data. only once per
 * iteration, if the correct conditions are met (i.e. indicating ready
 * to commit), it commits the next set of compressed bytes to the output 
 * destination. each commit appends an optional unencoded portion
 * followed by an encoded porition. state variables which control the 
 * commit are:
 *   - prefix - number of bytes which are encoded/represented
 *              by the unencoded portion of the next commit 
 *   - span   - number of bytes which are encoded by the
 *              encoded portion of the next commit
 * 
 * this is transactional in the sense that it may take N bytes of input
 * to read/process before the unencoded+encoded representation of those
 * bytes is actually commited into the destination. at the point of 
 * commit, N is equal to prefix+span. therefore the set of bytes from
 * the src which is handled by the commit are at prefix+span bytes
 * -behind- the current src pointer (i.e. src-(prefix+span)). prefix and
 * span are reset (to 0) after each commit so that a new transaction can
 * be built for the next set of input bytes.
 * 
 * decoding the committed bytes will thus result in the corresponding 
 * set of bytes from the src 
 * 
 * for the first 9 bytes of input, the prefix value is forcibly incremen-
 * ted by 1 (for each iteration) and nothing else is done. this prepares 
 * those bytes for contributing to the unencoded portion of the next 
 * commit, for (as explained momentarily) the encoded portion requires
 * at least 9 bytes of preceding data in the output stream
 * 
 * for all iterations after the 9th iteration, the byte of input 9 bytes 
 * behind the src is inserted into a suffix tree at the start of the 
 * iteration, and the current byte of input data (at src) is looked up
 * amongst the children of the previous byte of input data if it was
 * found as a child in the previous iteration and so on-and otherwise 
 * amongst the children of the root. starting at the 10th iteration, 
 * this means the longest 
 * 
 */


void page_sizes_verify(page_compressed *page_dst,
		       uint32_t src_offset, uint8_t *dst) {
  uint8_t *ptr;
  uint8_t span, data;
  uint16_t remainder;
  size_t size, compressed_size;
  
  size = 0;
  ptr = page_dst->data;
 // printf("offsets\n");
  while (size < src_offset) {
   // printf("%08x ", src - (uint8_t*)page_src);
    span = *(ptr++);
    if (span & 0x80) {
      data = *(ptr++);
      span = (data & 7) + 3; 
      // if (span == 10) span = 0x40;
    }
    else
      ptr += span;
    size += span;
  }  
  compressed_size = ptr - (uint8_t*)page_dst;
  
  printf("cur decompressed_size: %i\n", size);
  printf("cur src offset: %i\n", src_offset);
  printf("cur compressed size: %i\n", compressed_size);
  printf("cur dst offset: %i\n", dst - (uint8_t*)page_dst);
}
